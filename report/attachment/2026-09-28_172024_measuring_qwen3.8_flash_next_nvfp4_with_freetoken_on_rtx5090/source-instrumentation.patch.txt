+"""Benchmark-only host wall timers; no CUDA synchronization or per-token file I/O.
+
+Start: first prepared prefill admission. First/last token: scheduler publication,
+after its existing result synchronization. Prefill includes chunked scheduling,
+MTP catch-up and first-token sampling, but excludes tokenization and queue wait.
+Only a completed request is written. No prompts, tokens or generated text logged.
+"""
+import json
+import os
+import time
+
+_PATH = os.environ.get("FREETOKEN_BENCH_TIMING_FILE")
+_REQUESTS = {}
+
+
+def admitted(rows):
+    if not _PATH:
+        return
+    now = time.perf_counter()
+    for uid, prompt_tokens, cached_tokens in rows:
+        if uid in _REQUESTS:
+            raise RuntimeError(f"benchmark timing: repeated admission for {uid}")
+        _REQUESTS[uid] = dict(uid=uid, prompt_tokens=prompt_tokens,
+                              cached_tokens=cached_tokens, admitted_at=now,
+                              first_at=None, first_batch_tokens=0,
+                              completion_tokens=0)
+
+
+def published(messages):
+    if not _PATH or not messages:
+        return
+    now = time.perf_counter()
+    counts = {}
+    for msg in messages:
+        counts[msg.uid] = counts.get(msg.uid, 0) + 1
+    for uid, n in counts.items():
+        state = _REQUESTS[uid]
+        if state["first_at"] is None:
+            state["first_at"] = now
+            state["first_batch_tokens"] = n
+        state["completion_tokens"] += n
+    for msg in messages:
+        if not msg.finished:
+            continue
+        state = _REQUESTS.pop(msg.uid)
+        state["last_at"] = now
+        state["finish_reason"] = msg.finish_reason
+        state["prompt_n"] = state["prompt_tokens"] - state["cached_tokens"]
+        state["prefill_wall_s"] = state["first_at"] - state["admitted_at"]
+        state["decode_wall_s"] = state["last_at"] - state["first_at"]
+        state["prefill_tps"] = state["prompt_n"] / state["prefill_wall_s"]
+        state["decode_tps"] = (
+            (state["completion_tokens"] - state["first_batch_tokens"])
+            / state["decode_wall_s"] if state["decode_wall_s"] > 0 else None)
+        with open(_PATH, "a") as out:
+            out.write(json.dumps(state) + "\n")
